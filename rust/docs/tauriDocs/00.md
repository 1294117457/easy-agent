##### 1.tauri架构

```
capabilities、gen、icons、
src/lib.rs、src/main.rs
build.rs、Cargo.rs,Cargo.toml,tauri.conf.json

capabilities
	配置权限、开发时按需配置
gen
	运行或打包时生成的schema文件，自动补全和校验
icons
	配置应用图标
build.rs
	构建脚本，解析tauri.conf.json
	生成tauri运行时代码
tauri.conf.json
	应用配置，定义元信息、窗口配置、构建命令、打包设置
main.rs
	入口，调用lib.rs定义的run方法
lib.rs
	和前端连接的桥梁
	注册opener、暴露greet命令给前端、启动tauri（run方法）

│  ┌─────────────┐                               │
│  │  build.rs   │───调用 tauri_build::build()──→│
│  └─────────────┘                               │
│                            │                    │
│                            ▼                    │
│               ┌───────────────────┐             │
│               │  gen/schemas/*.json│ ← 自动生成  │
│               └───────────────────┘             │
│                                                  │
│  ┌─────────────────────────────────────────┐    │
│  │  lib.rs: Tauri 应用逻辑                  │    │

│  └─────────────────────────────────────────┘    │
└─────────────────────────────────────────────────┘
```

```
简单地说
tauri.conf.json被lib.rs读取，
```

```
impl SqliteStorageAdapter {
    pub fn new(db_path: &str) -> Result<Self, StorageError> {
        let conn = Connection::open(db_path)
            .map_err(|e| StorageError::ConnectionFailed(e.to_string()))?;
        Ok(Self { conn })
    }
}
impl StoragePort for SqliteStorageAdapter {
    fn init(&self) -> Result<(), StorageError> {
        self.conn.execute(
            "CREATE TABLE IF NOT EXISTS api_keys (
                id TEXT PRIMARY KEY,
                name TEXT NOT NULL,
                key TEXT NOT NULL,
                provider TEXT NOT NULL,
                status TEXT NOT NULL,
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL,
                last_used_at TEXT
            )",
            []
        ).map_err(|e| StorageError::QueryFailed(e.to_string()))?;
        Ok(())
    }
    fn save_api_key(&self, api_key: &super::domain::api_key::ApiKey) -> Result<(), StorageError> {
        self.conn.execute(
            "INSERT INTO api_keys (id, name, key, provider, status, created_at, updated_at, last_used_at)
             VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8)",
            params![
                api_key.id.0,
                api_key.name,
                api_key.key,
                serde_json::to_string(&api_key.provider).unwrap(),
                serde_json::to_string(&api_key.status).unwrap(),
                api_key.created_at.to_rfc3339(),
                api_key.updated_at.to_rfc3339(),
                api_key.last_used_at.map(|dt| dt.to_rfc3339()),
            ],
        ).map_err(|e| StorageError::QueryFailed(e.to_string()))?;
        Ok(())
    }
    fn get_api_key(&self, id: &str) -> Result<Option<super::domain::api_key::ApiKey>, StorageError> {
        let mut stmt = self.conn.prepare(
            "SELECT id, name, key, provider, status, created_at, updated_at, last_used_at FROM api_keys WHERE id = ?1"
        ).map_err(|e| StorageError::QueryFailed(e.to_string()))?;
        let result = stmt.query_row(params![id], |row| {
            // 这里需要从数据库行构建 ApiKey
            // 暂时留空，后面补全
            Ok(())
        });
        match result {
            Ok(api_key) => Ok(Some(api_key)),
            Err(rusqlite::Error::QueryReturnedNoRows) => Ok(None),
            Err(e) => Err(StorageError::QueryFailed(e.to_string())),
        }
    }

pub mod value_objects;
use serde::{Deserialize, Serialize};
use chrono::{DateTime, Utc};
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ApiKey {
    pub id: value_objects::ApiKeyId,
    pub name: String,
    pub key: String,
    pub provider: value_objects::LlmProvider,
    pub status: value_objects::KeyStatus,
    pub created_at: DateTime<Utc>,
    pub updated_at: DateTime<Utc>,
    pub last_used_at: Option<DateTime<Utc>>,
}
impl ApiKey {
    pub fn new(name: String, raw_key: String, provider: value_objects::LlmProvider) -> Self {
        Self {
            id: value_objects::ApiKeyId::new(),
            name,
            key: raw_key,
            provider,
            status: value_objects::KeyStatus::Active,
            created_at: Utc::now(),
            updated_at: Utc::now(),
            last_used_at: None,
        }
    }
    pub fn is_active(&self) -> bool {
        self.status == value_objects::KeyStatus::Active
    }
}
```
