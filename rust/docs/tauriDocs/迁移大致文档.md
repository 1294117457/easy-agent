## 1. 迁移概述

```
┌─────────────────────────────────────────────────────────────────┐
│                                                                 │
│   迁移目标                                                       │
│   ─────────────────────────────────────────────────────────     │
│                                                                 │
│   electron1 (Electron + TypeScript)                             │
│           │                                                      │
│           │ 逐步迁移                                              │
│           ▼                                                      │
│   tauri-easy-agent (Tauri + Vue + Rust)                         │
│                                                                 │
│   ┌─────────────────────────────────────────────────────────┐  │
│   │                                                         │  │
│   │   前端保持不变：Vue 3 + TypeScript + Pinia              │  │
│   │   ├── src/views/                                       │  │
│   │   ├── src/components/                                  │  │
│   │   ├── src/stores/                                     │  │
│   │   └── src/api/                                        │  │
│   │                                                         │  │
│   │   后端完全重写：Rust + Tauri                            │  │
│   │   ├── LLM 管理                                        │  │
│   │   ├── 工作流引擎                                       │  │
│   │   ├── SQLite 持久化                                   │  │
│   │   └── MCP 集成                                        │  │
│   │                                                         │  │
│   └─────────────────────────────────────────────────────────┘  │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

### 迁移原则

1. **前端不变**：Vue 代码直接迁移，保持功能一致
2. **后端重写**：Electron Main Process → Rust Tauri Backend
3. **IPC 重构**：`ipcMain.handle()` → `#[tauri::command]`
4. **分步验证**：每迁移一个模块，确保功能正常
5. **electron1 参考**：始终作为正确性参考

## 5. 迁移步骤详解

### 5.1 第一步：创建 Tauri 项目

```bash
# 进入项目目录
cd d:\XMU\3UP\交互设计\codeGithub\easy-agent

# 创建 Tauri 项目（使用 Vue + TypeScript 模板）
npm create tauri-app@latest tauri-easy-agent -- --template vue-ts

# 进入项目
cd tauri-easy-agent

# 安装前端依赖
npm install

# 安装 Tauri API
npm install @tauri-apps/api
```

use thiserror::Error;

#[derive(Error, Debug)]

use tauri::{State, Window};
use crate::state::AppState;
use crate::error::AppError;

pub async fn chat_send(
    state: State<'_, AppState>,
    window: Window,
    conversation_id: String,
    message: String,
) -> Result<(), String> {
    // 获取回调
    let llm_manager = state.llm_manager.clone();
    let window_clone = window.clone();

    // 创建流式回调
    let token_sender = move |token: String| {
        let _ = window_clone.emit("agent:token", token);
    };

    let done_sender = window.clone();
    let done_callback = move || {
        let _ = done_sender.emit("agent:done", ());
    };

    let error_callback = move |error: String| {
        let _ = error_sender.emit("agent:error", error);
    };

    // 调用业务逻辑
    llm_manager
        .send_message(&conversation_id, &message, token_sender, done_callback, error_callback)
        .await
        .map_err(|e| e.to_string())?;

    Ok(())
}

// src-tauri/src/services/workflow_service.rs

use std::collections::{HashMap, VecDeque};
use crate::domain::workflow::{Workflow, WorkflowEdge, WorkflowNode, WorkflowResult};

pub struct WorkflowService {
    workflows: HashMap<String, Workflow>,
    workflow_nodes: HashMap<String, Vec<WorkflowNode>>,
}

impl WorkflowService {
    pub fn new() -> Self {
        Self {
            workflows: HashMap::new(),
            workflow_nodes: HashMap::new(),
        }
    }

    pub fn create_workflow(&mut self, name: String, description: Option<String>) -> Workflow {
        let workflow = Workflow::new(name, description);
        self.workflows.insert(workflow.id.clone(), workflow.clone());
        self.workflow_nodes.insert(workflow.id.clone(), Vec::new());
        workflow
    }

    pub fn add_node(&mut self, workflow_id: &str, node: WorkflowNode) -> Result<(), AppError> {
        let nodes = self.workflow_nodes
            .get_mut(workflow_id)
            .ok_or_else(|| AppError::NotFound(format!("Workflow {} not found", workflow_id)))?;

        if !nodes.iter().any(|n| n.id == node.id) {
            nodes.push(node);
        }
        Ok(())
    }

    pub async fn execute(
        &self,
        workflow_id: &str,
        input: serde_json::Value,
    ) -> Result<WorkflowResult, AppError> {
        let workflow = self.workflows
            .get(workflow_id)
            .ok_or_else(|| AppError::NotFound(format!("Workflow {} not found", workflow_id)))?;

        let nodes = self.workflow_nodes
            .get(workflow_id)
            .ok_or_else(|| AppError::NotFound(format!("Workflow {} not found", workflow_id)))?;

        // 拓扑排序
        let execution_order = self.topological_sort(nodes, &workflow.edges)?;

        // 执行
        let mut context: HashMap<String, serde_json::Value> = HashMap::new();
        context.insert("__input__".to_string(), input);

        for node in execution_order {
            let node_input = self.get_node_input(&node.id, &workflow.edges, &context)?;
            let node_output = self.execute_node(&node, node_input).await?;
            context.insert(node.id.clone(), node_output);
        }

        Ok(WorkflowResult {
            success: true,
            output: context.get(&workflow.edges.last()
                .map(|e| e.target_node_id.clone())
                .unwrap_or_default())
                .cloned(),
            node_results: Some(context),
            error: None,
        })
    }

    fn topological_sort(
        &self,
        nodes: &[WorkflowNode],
        edges: &[WorkflowEdge],
    ) -> Result<Vec<WorkflowNode>, AppError> {
        let mut in_degree: HashMap<String, usize> = HashMap::new();
        let mut adjacency: HashMap<String, Vec<String>> = HashMap::new();

        for node in nodes {
            in_degree.insert(node.id.clone(), 0);
            adjacency.insert(node.id.clone(), Vec::new());
        }

        for edge in edges {
            adjacency.entry(edge.source_node_id.clone())
                .or_default()
                .push(edge.target_node_id.clone());
            *in_degree.entry(edge.target_node_id.clone()).or_insert(0) += 1;
        }

        let mut queue: VecDeque<String> = in_degree
            .iter()
            .filter(|(_, &degree)| degree == 0)
            .map(|(id, _)| id.clone())
            .collect();

        let mut result: Vec<WorkflowNode> = Vec::new();

        while let Some(node_id) = queue.pop_front() {
            if let Some(node) = nodes.iter().find(|n| n.id == node_id) {
                result.push(node.clone());
            }

            for neighbor in adjacency.get(&node_id).iter().flatten() {
                if let Some(degree) = in_degree.get_mut(neighbor) {
                    *degree -= 1;
                    if *degree == 0 {
                        queue.push_back(neighbor.clone());
                    }
                }
            }
        }

        if result.len() != nodes.len() {
            return Err(AppError::Validation("Cycle detected in workflow".into()));
        }

        Ok(result)
    }
}

use crate::domain::message::Message;

pub struct SqliteStorage {
    pool: SqlitePool,
}

impl SqliteStorage {
    pub async fn new(database_url: &str) -> Result<Self, AppError> {
        let pool = SqlitePool::connect(database_url).await?;
        Ok(Self { pool })
    }

    pub async fn init_schema(&self) -> Result<(), AppError> {
        sqlx::query(
            r#"
            CREATE TABLE IF NOT EXISTS messages (
                id TEXT PRIMARY KEY,
                conversation_id TEXT NOT NULL,
                role TEXT NOT NULL,
                content TEXT NOT NULL,
                model TEXT,
                created_at TEXT NOT NULL,
                is_compressed INTEGER NOT NULL DEFAULT 0
            )
            "#,
        )
        .execute(&self.pool)
        .await?;

        Ok(())
    }

    pub async fn save_message(&self, message: &Message) -> Result<(), AppError> {
        sqlx::query(
            r#"
            INSERT INTO messages (id, conversation_id, role, content, model, created_at, is_compressed)
            VALUES (?, ?, ?, ?, ?, ?, ?)
            "#,
        )
        .bind(&message.id)
        .bind(&message.conversation_id)
        .bind(serde_json::to_string(&message.role)?)
        .bind(&message.content)
        .bind(&message.model)
        .bind(message.created_at.to_rfc3339())
        .bind(message.is_compressed as i32)
        .execute(&self.pool)
        .await?;

        Ok(())
    }

    pub async fn get_messages(&self, conversation_id: &str) -> Result<Vec<Message>, AppError> {
        let rows = sqlx::query(
            "SELECT * FROM messages WHERE conversation_id = ? ORDER BY created_at ASC"
        )
        .bind(conversation_id)
        .fetch_all(&self.pool)
        .await?;

        let messages = rows.iter().map(|row| {
            Message {
                id: row.get("id"),
                conversation_id: row.get("conversation_id"),
                role: serde_json::from_str(&row.get::<String, _>("role")).unwrap_or(MessageRole::User),
                content: row.get("content"),
                model: row.get("model"),
                created_at: chrono::DateTime::parse_from_rfc3339(&row.get::<String, _>("created_at"))
                    .map(|dt| dt.with_timezone(&chrono::Utc))
                    .unwrap_or_else(|_| chrono::Utc::now()),
                is_compressed: row.get::<i32, _>("is_compressed") != 0,
            }
        }).collect();

        Ok(messages)
    }
