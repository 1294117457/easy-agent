##### 1聊天相关模块

```
apikey、prompt、model、config、conversation
	conversation聚合message
按功能驱动
    对话的trait属于model，model管理对话的功能
    apikey、prompt、skill管理自身状态；
    config配置当前选择的上述apikey、prompt、skill，供model对话时使用
    conversation聚合message，管理上下文对话
```

| 设计点      | 方案                     | 理由                 |
| :---------- | :----------------------- | :------------------- |
| 资源存储    | ApiKey/Prompt/Skill 实体 | 各自管理状态         |
| 资源引用    | Config 直接持有          | 利用所有权转移       |
| Config 共享 | `Arc<Config>`            | 避免所有权冲突       |
| Model 定义  | trait + Adapter          | 多 Provider 统一接口 |
| 调用入口    | LlmService               | 统一资源管理         |

    enabled: bool,

// Config 直接持有资源，所有权在这里聚合
struct Config {
    name: String,
    
    // 资源
    api_key: ApiKey,          // 直接持有，不再是 ID
    prompt: Option<Prompt>,    // 可选
    skills: Vec<Skill>,       // 可选
    
    // 运行时参数
    model_name: String,
    temperature: f32,
    max_tokens: u32,
}

impl Config {
    // 构建最终的 system prompt（融合 prompt + skills）
    pub fn build_system_prompt(&self) -> String {
        let mut prompt = String::new();
        
        if let Some(p) = &self.prompt {
            prompt.push_str(&p.content);
            prompt.push_str("\n\n");
        }
        
        if !self.skills.is_empty() {
            prompt.push_str("Available skills:\n");
            for skill in &self.skills {
                prompt.push_str(&format!("- {}: {}\n", skill.name, skill.commands.join(", ")));
            }
        }
        
        prompt
    }
}

struct Conversation {
    id: ConversationId,
    config: Config,           // 持有 Config 的所有权
    messages: Vec<Message>,
    model: Arc<dyn Model>,    // 使用 Arc 共享 Model 实现
}

impl Conversation {
    pub async fn send(&mut self, content: String) -> Result<Message> {
        // 1. 创建用户消息
        let user_msg = Message::user(content);
        self.messages.push(user_msg.clone());
        
        // 2. 调用 Model，Config 所有权转移
        let config = std::mem::take(&mut self.config)
            .expect("Config should be present");
        
        let response = self.model.chat(self.messages.clone(), config).await?;
        
        // 3. 恢复 Config（或保留给下次使用）
        // 注意：这里需要设计如何处理 Config 的所有权
        // 方案 A：每次 send 都传 Config
        // 方案 B：Config 不可变，Conversation 只引用
        
        // 4. 保存响应
        self.messages.push(response.clone());
        Ok(response)
    }
}
